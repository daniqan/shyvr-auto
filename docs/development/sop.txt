# Simple RL Agent (Policy Gradient)
class SimpleRLAgent:
    def __init__(self, num_features: int, num_actions: int, learning_rate: float = 0.01):
        self.weights = np.zeros((num_actions, num_features + 1))  # +1 for bias
        self.lr = learning_rate

    def softmax(self, x: np.ndarray) -> np.ndarray:
        e_x = np.exp(x - np.max(x))
        return e_x / e_x.sum()

    def predict(self, state: np.ndarray) -> int:
        state = np.append(state, 1)  # Add bias
        logits = np.dot(self.weights, state)
        probs = self.softmax(logits)
        return np.random.choice(len(probs), p=probs)

    def update(self, state: np.ndarray, action: int, reward: float):
        state = np.append(state, 1)  # Add bias
        logits = np.dot(self.weights, state)
        probs = self.softmax(logits)
        grad = np.outer(probs, state)
        grad[action] -= state
        self.weights -= self.lr * reward * grad  # REINFORCE update (flip sign for negative rewards)

In the context of our Solana token deploy monitoring bot, a Q-learning greedy scheme would adapt the reinforcement learning (RL) component to use Q-learning, a model-free, off-policy algorithm. This is an improvement over the simple policy gradient approach in the previous code, as Q-learning explicitly estimates the value (Q-value) of each state-action pair, allowing for more stable learning and better handling of delayed rewards (e.g., waiting for a token's price to pump after a "wait" action).
What a Q-Learning Greedy Scheme Looks Like
Q-learning maintains a Q-table (or function approximator) that stores the expected long-term reward for each state-action pair. The agent learns by updating Q-values based on observed rewards and the Bellman equation:
$ Q(s, a) \leftarrow Q(s, a) + \alpha \left[ r + \gamma \max_{a'} Q(s', a') - Q(s, a) \right] $

$ s $: Current state (e.g., token features like normalized LP size, holder count, volume delta, sentiment score).
$ a $: Action taken (e.g., 0=skip, 1=buy_now, 2=wait_5min, 3=wait_10min).
$ r $: Immediate reward (e.g., profit % from the trade, 0 for skip, negative for losses/rugs).
$ s' $: Next state (in a sequential env; for episodic trading, this could be the next token candidate).
$ \alpha $: Learning rate (e.g., 0.1, controls update speed).
$ \gamma $: Discount factor (e.g., 0.99, values future rewards).
$ \max_{a'} Q(s', a') $: Best estimated value in the next state (off-policy aspect).

A greedy scheme means the agent always selects the action with the highest Q-value in a given state (pure exploitation: $ a = \arg\max_a Q(s, a) $). To balance exploration (trying new actions to learn better), we often use epsilon-greedy: With probability $\epsilon$ (e.g., 0.1, decaying over time), pick a random action; otherwise, be greedy. This prevents getting stuck in suboptimal policies.
In practice:

During training/backtesting: Use epsilon-greedy to explore.
In live deployment (greedy mode): Set $\epsilon = 0$ for exploitation, making decisions like "buy_now if Q-value > threshold" for high-confidence pumps.
Reinforcement loop: After each action, track the real outcome (e.g., wait 10min, measure profit), update Q-values, and retrain periodically on accumulated data.

This scheme shines for our bot because it learns timing (e.g., "wait_10min" might yield higher Q if volume is rising slowly) and skips rugs, improving over random or baseline buying.

        self.profits = self.profits * np.random.uniform(0.5, 1.5, num_samples)
        rugs = np.random.choice([True, False], num_samples, p=[0.1, 0.9])
        self.profits[rugs] = -0.7
        
        self.num_states = 10  # Discretize each feature into 10 bins for Q-table
        self.num_actions = 4  # 0=skip, 1=buy_now, 2=wait_5, 3=wait_10
        self.current_step = 0
        self.num_samples = num_samples
    
    def reset(self):
        self.current_step = 0
        return self.get_state(0)
    
    def get_state(self, step):
        # Normalize and discretize features
        lp_norm = min(self.lp_usds[step] / 100000, 1.0)
        holders_norm = min(self.holders[step] / 100, 1.0)
        vol_norm = min(max(self.volume_deltas[step], -1), 1)
        sent_norm = self.sentiments[step]
        
        # Discretize to 0-9
        state = (
            int(lp_norm * 9),
            int(holders_norm * 9),
            int((vol_norm + 1) / 2 * 9),  # Shift -1 to 1 to 0-1
            int((sent_norm - 0.4) / 0.5 * 9)  # 0.4-0.9 to 0-9
        )
        return state
    
    def step(self, action):
        profit = self.profits[self.current_step]
        if action == 0:  # skip
            reward = 0
        else:
            # Adjust for wait actions
            adjust = 1 - 0.05 * (action - 1)
            reward = profit * adjust
        
        self.current_step += 1
        done = self.current_step >= self.num_samples
        next_state = self.get_state(self.current_step) if not done else None
        return next_state, reward, done

# Q-Learning Agent
class QLearningAgent:
    def __init__(self, num_states, num_actions, alpha=0.1, gamma=0.99, epsilon=0.1):
        self.q_table = np.zeros((num_states, num_states, num_states, num_states, num_actions))  # 10^4 states
        self.alpha = alpha
        self.gamma = gamma
        self.epsilon = epsilon
        self.num_actions = num_actions
    
    def choose_action(self, state, greedy=False):
        if np.random.uniform(0, 1) < self.epsilon and not greedy:
            return np.random.randint(0, self.num_actions)
        else:
            return np.argmax(self.q_table[state])
    
    def update(self, state, action, reward, next_state):
        if next_state is not None:
            max_next_q = np.max(self.q_table[next_state])
            target = reward + self.gamma * max_next_q
        else:
            target = reward
        current_q = self.q_table[state][action]
        self.q_table[state][action] = current_q + self.alpha * (target - current_q)
